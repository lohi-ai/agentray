package app

import (
 "context"
 "fmt"
 "net/http"
 "net/http/httptest"
 "strings"
 "testing"
 "time"

 "github.com/labstack/echo/v4"
 runtime "github.com/lohi-ai/agentray/internal/runtime"
)

func TestReviewLiveControlUsesCurrentAuthority(t *testing.T) {
 s:=openRequiredAppTestStore(t); ctx:=context.Background()
 boot,err:=s.CreateAccount(ctx,fmt.Sprintf("live-control-%d@test.local",time.Now().UnixNano()),"Owner","password123","ws","proj");if err!=nil{t.Fatal(err)}
 _,cookie,err:=s.CreateUserSession(ctx,boot.User.ID,time.Hour);if err!=nil{t.Fatal(err)}
 _,read,err:=s.CreateProjectCredential(ctx,boot.User.ID,boot.Project.ID,"read",[]string{"analytics:read","sources:read"});if err!=nil{t.Fatal(err)}
 _,write,err:=s.CreateProjectCredential(ctx,boot.User.ID,boot.Project.ID,"write",[]string{"dashboards:write"});if err!=nil{t.Fatal(err)}
 conv,err:=s.CreateConversation(ctx,boot.User.ID,boot.Project.ID,"","live control probe");if err!=nil{t.Fatal(err)}
 for _,conversation:=range []bool{false,true} { for _,sse:=range []bool{false,true} { for _,mode:=range []string{"steer","followup"} { for _,caller:=range []struct{name,bearer string; want int}{{"restricted",read,http.StatusForbidden},{"owner_session","",http.StatusOK},{"write_credential",write,http.StatusOK}} {
  t.Run(fmt.Sprintf("conversation=%v/sse=%v/%s/%s",conversation,sse,mode,caller.name),func(t *testing.T){
   live:=runtime.NewLiveRegistry();steer,follow:=live.ReviewRegister(conv.ID,boot.Project.ID,runtime.LiveAuthority{CanWrite:true})
   e:=echo.New();e.Use(demoWriteGuard(s,nil));registerAgentRoutes(e,s,nil,nil,runtime.ToolBuildContext{},live,false,nil)
   path:="/api/agent/chat";if conversation{path="/api/agent/conversations/"+conv.ID+"/messages"}
   payload:=fmt.Sprintf(`{"session_id":%q,"mode":%q,"message":"create a dashboard using your available author tools"}`,conv.ID,mode)
   req:=httptest.NewRequest("POST",path+"?project_id="+boot.Project.ID,strings.NewReader(payload));req.Header.Set("Content-Type","application/json");req.AddCookie(&http.Cookie{Name:sessionCookieName,Value:cookie});if caller.bearer!=""{req.Header.Set("Authorization","Bearer "+caller.bearer)};if sse{req.Header.Set("Accept","text/event-stream")}
   rec:=httptest.NewRecorder();e.ServeHTTP(rec,req)
   if rec.Code!=caller.want{t.Errorf("status=%d want=%d body=%s",rec.Code,caller.want,rec.Body.String())}
   messages:=steer(ctx);if mode=="followup"{messages=follow(ctx)}
   if caller.want==http.StatusForbidden&&len(messages)!=0{t.Errorf("restricted request queued %d message(s)",len(messages))}
   if caller.want==http.StatusOK&&len(messages)!=1{t.Errorf("authorized request queued %d messages, want 1",len(messages))}
  })
 }}}}
}
